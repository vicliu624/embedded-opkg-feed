cmake_minimum_required(VERSION 3.16)
project(tdvp_onnxruntime_shared_registry_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ONNX CONFIG REQUIRED)
if(NOT EXISTS "${TDVP_ORT_LIBRARY}" OR NOT EXISTS "${TDVP_ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Actual target ONNX Runtime library and API headers are required")
endif()
add_library(tdvp_ort SHARED IMPORTED)
set_target_properties(tdvp_ort PROPERTIES IMPORTED_LOCATION "${TDVP_ORT_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${TDVP_ORT_INCLUDE_DIR}")
add_executable(ai-onnxruntime-shared-registry-smoke ../../ai-onnxruntime-shared-registry-smoke.cpp)
target_link_libraries(ai-onnxruntime-shared-registry-smoke PRIVATE ONNX::onnx tdvp_ort)
